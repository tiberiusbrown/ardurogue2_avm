add_library(ardurogue2_sim_core STATIC
    simulator.cpp omniscient_agent.cpp metrics.cpp trace.cpp frontend.cpp experiment.cpp
    ../src/state.cpp ../src/combat.cpp ../src/combat_math.cpp ../src/items.cpp
    ../src/world.cpp ../src/world_gen.cpp ../src/world_gen_population.cpp)
target_compile_features(ardurogue2_sim_core PUBLIC cxx_std_17)
target_compile_definitions(ardurogue2_sim_core PUBLIC ARDUROGUE2_SIM)
target_include_directories(ardurogue2_sim_core PUBLIC . ../src)
if(MSVC)
    target_compile_options(ardurogue2_sim_core PRIVATE /W4)
else()
    target_compile_options(ardurogue2_sim_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
set(SIM_GIT_SHA "unknown")
set(SIM_GIT_DIRTY "\"unknown\"")
set(SIM_BUILD_TYPE "unknown")
if(CMAKE_BUILD_TYPE)
    set(SIM_BUILD_TYPE "${CMAKE_BUILD_TYPE}")
endif()
set(SIM_COMPILER_ID "unknown")
if(CMAKE_CXX_COMPILER_ID)
    set(SIM_COMPILER_ID "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
endif()
find_package(Git QUIET)
if(GIT_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${PROJECT_SOURCE_DIR}" rev-parse HEAD
        WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" OUTPUT_VARIABLE SIM_REV OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE SIM_REV_RESULT)
    if(SIM_REV_RESULT EQUAL 0)
        set(SIM_GIT_SHA "${SIM_REV}")
        execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${PROJECT_SOURCE_DIR}" status --porcelain
            WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" OUTPUT_VARIABLE SIM_STATUS ERROR_QUIET RESULT_VARIABLE SIM_STATUS_RESULT)
        if(SIM_STATUS_RESULT EQUAL 0)
            set(SIM_GIT_DIRTY "false")
            if(SIM_STATUS)
                set(SIM_GIT_DIRTY "true")
            endif()
        endif()
    endif()
endif()
configure_file(build_metadata.hpp.in build_metadata.hpp @ONLY)
add_executable(ardurogue2_sim main.cpp parallel.cpp manifest.cpp)
target_include_directories(ardurogue2_sim PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ardurogue2_sim PRIVATE ardurogue2_sim_core)
add_executable(sim_checks tests.cpp policy_checks.cpp)
target_link_libraries(sim_checks PRIVATE ardurogue2_sim_core)
add_test(NAME simulator_checks COMMAND sim_checks)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME simulator_cli COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/test_cli.py" "$<TARGET_FILE:ardurogue2_sim>")
endif()
set(BALANCE_PYTHON "${Python3_EXECUTABLE}" CACHE FILEPATH "Optional Python with balance statistical dependencies")
if(BALANCE_PYTHON)
    execute_process(COMMAND "${BALANCE_PYTHON}" -c "import numpy, pandas, scipy, statsmodels" RESULT_VARIABLE BALANCE_IMPORT ERROR_QUIET)
    if(BALANCE_IMPORT EQUAL 0)
        add_test(NAME balance_statistics COMMAND "${BALANCE_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/test_balance.py")
        add_test(NAME balance_cli COMMAND "${BALANCE_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/test_balance_cli.py" "$<TARGET_FILE:ardurogue2_sim>")
    endif()
endif()
